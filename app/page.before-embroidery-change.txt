/* =========================================================
   TECHCRAFT GENERATIVE HERO BACKGROUND
   ========================================================= */

function TechCraftHeroBackground() {
  const canvasRef = useRef<HTMLCanvasElement>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;

    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    let animationFrame = 0;
    let width = 0;
    let height = 0;

    const mouse = {
      x: -1000,
      y: -1000,
    };

    const points: {
      x: number;
      y: number;
      vx: number;
      vy: number;
      size: number;
    }[] = [];

    const resize = () => {
      const dpr = Math.min(window.devicePixelRatio || 1, 2);

      width = canvas.clientWidth;
      height = canvas.clientHeight;

      canvas.width = width * dpr;
      canvas.height = height * dpr;

      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

      points.length = 0;

      const count = Math.min(
        100,
        Math.max(45, Math.floor((width * height) / 16000))
      );

      for (let i = 0; i < count; i++) {
        points.push({
          x: Math.random() * width,
          y: Math.random() * height,
          vx: (Math.random() - 0.5) * 0.22,
          vy: (Math.random() - 0.5) * 0.22,
          size: Math.random() * 1.8 + 0.5,
        });
      }
    };

    const handleMouseMove = (event: MouseEvent) => {
      const rect = canvas.getBoundingClientRect();

      mouse.x = event.clientX - rect.left;
      mouse.y = event.clientY - rect.top;
    };

    const handleMouseLeave = () => {
      mouse.x = -1000;
      mouse.y = -1000;
    };

    const drawGrid = (time: number) => {
      const spacing = 64;
      const offset = (time * 0.012) % spacing;

      ctx.save();

      ctx.lineWidth = 0.5;
      ctx.strokeStyle = "rgba(0, 66, 149, 0.14)";

      for (let x = -spacing + offset; x < width + spacing; x += spacing) {
        ctx.beginPath();
        ctx.moveTo(x, 0);
        ctx.lineTo(x, height);
        ctx.stroke();
      }

      for (let y = -spacing + offset; y < height + spacing; y += spacing) {
        ctx.beginPath();
        ctx.moveTo(0, y);
        ctx.lineTo(width, y);
        ctx.stroke();
      }

      ctx.restore();
    };

    const drawEmbroideryGeometry = (time: number) => {
      const centerX = width * 0.78;
      const centerY = height * 0.50;

      const pulse = Math.sin(time * 0.0007) * 10;

      ctx.save();

      ctx.translate(centerX, centerY);
      ctx.rotate(Math.sin(time * 0.00012) * 0.035);

      ctx.strokeStyle = "rgba(0, 66, 149, 0.25)";
      ctx.lineWidth = 1;

      const sizes = [
        125 + pulse,
        195 + pulse,
        275 + pulse,
        355 + pulse,
      ];

      sizes.forEach((size, index) => {
        ctx.beginPath();

        ctx.moveTo(0, -size);
        ctx.lineTo(size, 0);
        ctx.lineTo(0, size);
        ctx.lineTo(-size, 0);
        ctx.closePath();

        ctx.globalAlpha = 0.65 - index * 0.10;
        ctx.stroke();
      });

      // Inner diamond stitching
      for (let i = 0; i < 8; i++) {
        const angle = (Math.PI * 2 * i) / 8;
        const radius = 72 + Math.sin(time * 0.001 + i) * 5;

        const x = Math.cos(angle) * radius;
        const y = Math.sin(angle) * radius;

        ctx.beginPath();
        ctx.moveTo(x, y);
        ctx.lineTo(
          Math.cos(angle + 0.35) * (radius + 18),
          Math.sin(angle + 0.35) * (radius + 18)
        );
        ctx.stroke();
      }

      ctx.restore();
    };

    const drawConnections = () => {
      for (let i = 0; i < points.length; i++) {
        const p = points[i];

        for (let j = i + 1; j < points.length; j++) {
          const q = points[j];

          const dx = p.x - q.x;
          const dy = p.y - q.y;

          const distance = Math.sqrt(dx * dx + dy * dy);

          if (distance < 120) {
            const opacity = (1 - distance / 120) * 0.24;

            ctx.strokeStyle = `rgba(0, 66, 149, ${opacity})`;
            ctx.lineWidth = 0.6;

            ctx.beginPath();
            ctx.moveTo(p.x, p.y);
            ctx.lineTo(q.x, q.y);
            ctx.stroke();
          }
        }
      }
    };

    const drawMouseField = () => {
      if (mouse.x < 0) return;

      const radius = 180;

      const gradient = ctx.createRadialGradient(
        mouse.x,
        mouse.y,
        0,
        mouse.x,
        mouse.y,
        radius
      );

      gradient.addColorStop(0, "rgba(0, 66, 149, 0.20)");
      gradient.addColorStop(1, "rgba(0, 66, 149, 0)");

      ctx.fillStyle = gradient;

      ctx.beginPath();
      ctx.arc(mouse.x, mouse.y, radius, 0, Math.PI * 2);
      ctx.fill();
    };

    const animate = (time: number) => {
      ctx.clearRect(0, 0, width, height);

      drawGrid(time);
      drawEmbroideryGeometry(time);

      points.forEach((point) => {
        point.x += point.vx;
        point.y += point.vy;

        if (point.x < -20) point.x = width + 20;
        if (point.x > width + 20) point.x = -20;

        if (point.y < -20) point.y = height + 20;
        if (point.y > height + 20) point.y = -20;
      });

      drawConnections();

      points.forEach((point) => {
        ctx.beginPath();

        ctx.fillStyle = "rgba(0, 66, 149, 0.52)";

        ctx.arc(
          point.x,
          point.y,
          point.size,
          0,
          Math.PI * 2
        );

        ctx.fill();
      });

      drawMouseField();

      animationFrame = requestAnimationFrame(animate);
    };

    resize();

    window.addEventListener("resize", resize);
    window.addEventListener("mousemove", handleMouseMove);
    window.addEventListener("mouseleave", handleMouseLeave);

    animationFrame = requestAnimationFrame(animate);

    return () => {
      cancelAnimationFrame(animationFrame);

      window.removeEventListener("resize", resize);
      window.removeEventListener("mousemove", handleMouseMove);
      window.removeEventListener("mouseleave", handleMouseLeave);
    };
  }, []);

  return (
    <canvas
      ref={canvasRef}
      className="techcraft-generative-background"
      aria-hidden="true"
    />
  );
}

/* =========================================================
   POINTER
   ========================================================= */

function MagneticPointer() {
  const pointerRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const move = (event: MouseEvent) => {
      if (!pointerRef.current) return;

      pointerRef.current.style.transform = `translate3d(${event.clientX}px, ${event.clientY}px, 0)`;
    };

    window.addEventListener("mousemove", move);

    return () => window.removeEventListener("mousemove", move);
  }, []);

  return <div ref={pointerRef} className="pointer" />;
}
